   - Escape sequences supported
   - Syntax: "Hello ${name}, you are ${age} years old"

       EssentialLexer,
       EssentialParser,
       EssentialInterpreter,
       ModuleLoader
   )

   # Parse and execute
   lexer = EssentialLexer(source_code)
   tokens = lexer.tokenize()
   parser = EssentialParser(tokens)
   ast = parser.parse()

   interpreter = EssentialInterpreter()
   interpreter.execute(ast)
