USAGE EXAMPLES
============================================================================

Example 1: File Processing
--------------------------
import { read_lines, write_lines } from "files.lament"
import { map, filter } from "core.lament"
import { upper, strip } from "strings.lament"

remember lines = read_lines("input.txt")
sigh process(line) { exhale upper(strip(line)) }
remember processed = map(process, lines)
write_lines("output.txt", processed)


Example 2: HTTP API Client
---------------------------
import { http_get, http_post, parse_json } from "network.lament"

remember response = http_get("https://api.example.com/users")
remember users = parse_json(response["body"])

remember new_user = {"name": "Alice", "email": "alice@example.com"}
remember result = http_post("https://api.example.com/users", new_user)


Example 3: Data Analysis
-------------------------
import { mean, median, stdev } from "math.lament"
import { Counter_new } from "collections.lament"

remember data = [23, 45, 12, 67, 34, 23, 56]
confess "Mean: ${mean(data)}"
confess "Median: ${median(data)}"
confess "Std Dev: ${stdev(data)}"

remember counter = Counter_new(data)
confess counter["most_common"](3)


Example 4: Queue Processing
----------------------------
import { Queue_new } from "collections.lament"

remember queue = Queue_new()
queue["enqueue"]("task1")
queue["enqueue"]("task2")
queue["enqueue"]("task3")

while not queue["is_empty"]() {
    remember task = queue["dequeue"]()
    confess "Processing: ${task}"
}


Example 5: Cryptography
------------------------
import { hash_password, verify_password, uuid } from "crypto.lament"

remember hash_info = hash_password("my_password")
remember is_valid = verify_password("my_password", hash_info)

remember token = uuid()
confess "Generated token: ${token}"


Example 6: Date Manipulation
-----------------------------
import { today, add_days, format_date } from "datetime.lament"

remember dt = today()
remember next_week = add_days(dt, 7)
remember formatted = format_date(next_week, "%B %d, %Y")
confess "Next week: ${formatted}"
